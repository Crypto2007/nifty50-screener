from concurrent.futures import ThreadPoolExecutor, as_completed
import json
import os
import io
import requests
import pandas as pd
import yfinance as yf


# --- DYNAMIC INDEX SCRAPING (100% LIVE) ---

def get_live_nifty50_tickers() -> list:
    """Dynamically fetches current Nifty 50 constituents live from official sources."""
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
    }

    # Approach A: Fetch Wikipedia Table
    try:
        url = "https://en.wikipedia.org/wiki/NIFTY_50"
        response = requests.get(url, headers=headers, timeout=10)
        tables = pd.read_html(io.StringIO(response.text))
        df = tables[2]  # Constituents table
        return [f"{symbol}.NS" for symbol in df["Symbol"].tolist()]
    except Exception as e:
        print(f"Wikipedia scrape failed: {e}. Trying secondary NSE endpoint...")

    # Approach B: Direct NSE India Official Index CSV
    try:
        nse_url = "https://archives.nseindia.com/content/indices/ind_nifty50list.csv"
        response = requests.get(nse_url, headers=headers, timeout=10)
        df = pd.read_csv(io.StringIO(response.text))
        return [f"{symbol}.NS" for symbol in df["Symbol"].tolist()]
    except Exception as e:
        print(f"NSE official CSV fetch failed: {e}")
        return []


def load_universe(universe_name: str = "nifty_50", file_path: str = "tickers.json") -> list:
    """
    Checks local JSON cache. If missing/empty, dynamically fetches 
    live tickers and caches them to avoid repeated network hits.
    """
    # Load from cache if it exists and isn't empty
    if os.path.exists(file_path):
        try:
            with open(file_path, "r") as f:
                data = json.load(f)
                tickers = data.get(universe_name, [])
                if tickers:
                    return tickers
        except Exception:
            pass  # If cache is corrupted, re-fetch

    # Otherwise, fetch live and populate cache
    print("Fetching live Nifty 50 constituents from web...")
    live_tickers = get_live_nifty50_tickers()

    if live_tickers:
        with open(file_path, "w") as f:
            json.dump({universe_name: live_tickers}, f, indent=2)
        print(f"Successfully cached {len(live_tickers)} tickers in '{file_path}'.")

    return live_tickers


# --- METRIC NORMALISATION & ERROR HANDLING ---

def format_compact_number(number: float) -> str:
    if number is None or not isinstance(number, (int, float)):
        return "N/A"
    if abs(number) >= 1e12:
        return f"{round(number / 1e12, 2)}T"
    elif abs(number) >= 1e9:
        return f"{round(number / 1e9, 2)}B"
    elif abs(number) >= 1e6:
        return f"{round(number / 1e6, 2)}M"
    return str(round(number, 2))


def safe_float(value, scale: float = 1.0, default=None):
    if value is None:
        return default
    try:
        val = float(value)
        return round(val * scale, 4)
    except (ValueError, TypeError):
        return default


def fetch_metrics(symbol: str) -> dict:
    try:
        ticker = yf.Ticker(symbol)
        info = ticker.info or {}

        raw_div = info.get("dividendYield")
        if raw_div is not None and isinstance(raw_div, (int, float)):
            div_val = raw_div * 100 if raw_div < 0.1 else raw_div
            div_yield_pct = round(div_val, 2)
            formatted_div = f"{div_yield_pct}%"
        else:
            div_yield_pct = 0.0
            formatted_div = "N/A"

        return {
            "Symbol": symbol.upper(),
            "Price": safe_float(info.get("currentPrice")),
            "Book Value": safe_float(info.get("bookValue")),
            "Trailing P/E": safe_float(info.get("trailingPE")),
            "Forward P/E": safe_float(info.get("forwardPE")),
            "Price to Book": safe_float(info.get("priceToBook")),
            "ROE": safe_float(info.get("returnOnEquity")),
            "Profit Margin": safe_float(info.get("profitMargins")),
            "Revenue Growth": safe_float(info.get("revenueGrowth")),
            "Dividend Yield Pct": div_yield_pct,
            "Dividend Yield": formatted_div,
            "Market Cap Raw": safe_float(info.get("marketCap")),
            "Market Cap": format_compact_number(info.get("marketCap"))
        }
    except Exception as e:
        print(f"[Error] Skipped {symbol}: {e}")
        return None


# --- MULTI-THREADED EXECUTOR ---

def fetch_universe_batch(tickers: list, max_workers: int = 10) -> list:
    results = []
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        future_to_symbol = {
            executor.submit(fetch_metrics, symbol): symbol 
            for symbol in tickers
        }
        for future in as_completed(future_to_symbol):
            symbol = future_to_symbol[future]
            try:
                data = future.result()
                if data:
                    results.append(data)
            except Exception as e:
                print(f"Failed thread execution for {symbol}: {e}")
    return results
